How to Install CustomerPulse on Any Website

Installing CustomerPulse on your store, Next.js app, or Shopify takes 10 seconds. Just copy and paste this script tag into the <head> section of your HTML: 

<!-- CustomerPulse Real-Time Retention Tracker -->
<script src="http://localhost:3000/tracker.js" data-site-id="my_store_01"></script>

What does it track automatically?

  • Page views & Dwell time: Understand what products and pages customers are browsing.
  • Time spent: Know immediately if a customer is stuck on checkout.
  • Cart abandonment: Detect when someone leaves with items still in their cart.

1. Keyed Customer Accounts (Primary Key Enforcement)

CREATE MATERIALIZED TABLE accounts_keyed (
  user_id STRING NOT NULL,
  account_name STRING,
  tier STRING,
  mrr DOUBLE,
  email STRING,
  PRIMARY KEY (user_id) NOT ENFORCED
) AS
SELECT
  COALESCE(user_id, '') AS user_id,
  account_name, tier, mrr, email
FROM customer_accounts;

2. Behavioral Friction Scoring (Windowed Stream Aggregation)

CREATE MATERIALIZED TABLE session_friction_scores AS
SELECT
  session_id, user_id,
  window_end AS calculated_at,
  COUNT(*) AS total_interactions,
  SUM(CASE
    WHEN event_type = 'payment_failed' THEN 45
    WHEN event_type = 'rage_click' THEN 25
    WHEN event_type = 'support_query' THEN 20
    WHEN event_type = 'checkout_start' AND cart_value > 500 THEN 15
    ELSE 1
  END) AS friction_score,
  MAX(cart_value) AS exposed_cart_value,
  MAX(page_path) AS last_active_page
FROM TABLE(
  TUMBLE(TABLE customer_clickstream, DESCRIPTOR(event_time), INTERVAL '1' MINUTE)
)
GROUP BY session_id, user_id, window_start, window_end;

3. Automated Retention Interventions (Temporal Table Join & Action Routing)

CREATE MATERIALIZED TABLE retention_interventions AS
SELECT
  f.session_id, f.user_id,
  a.account_name, a.tier, a.mrr,
  f.friction_score, f.exposed_cart_value,
  f.last_active_page, f.calculated_at,
  CASE
    WHEN a.tier = 'Enterprise' AND f.friction_score >= 40 THEN 'VIP_CSM_SLACK_ALERT'
    WHEN f.exposed_cart_value >= 1000 AND f.friction_score >= 40 THEN 'RESCUE_PROMO_15PCT'
    WHEN f.friction_score >= 30 THEN 'PROACTIVE_CHAT_OFFER'
    ELSE 'STANDARD_MONITORING'
  END AS recommended_action
FROM session_friction_scores f
JOIN accounts_keyed FOR SYSTEM_TIME AS OF f.calculated_at AS a
  ON f.user_id = a.user_id
WHERE f.friction_score >= 30;
